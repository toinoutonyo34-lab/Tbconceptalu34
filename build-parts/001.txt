function structuredData(type, data = {}) {
  const base = {
    '@context': 'https://schema.org',
    '@type': type,
    name: site.name,
    url: site.baseUrl,
    telephone: site.phoneHref,
    email: site.email,
    areaServed: { '@type': 'AdministrativeArea', name: 'Hérault' }
  };
  return `<script type="application/ld+json">${json({ ...base, ...data })}</script>`;
}
